\begin{recipe}{Panini caprese}
\label{recipe:panini-caprese}
\kicker[Hoofdgerecht,Vegetarisch,Italiaans,Lunch]{Hoofdgerecht · vegetarisch · Italiaans · lunch}
\index[register]{Panini caprese}
\index[register]{Mozzarella}
\index[register]{Tomaat}
\index[register]{Focaccia}

\meta{15 minuten \dvd Contactgrill}

\marginimage{images/panini-caprese}
\lettrine{D}{eze} panini caprese aten we voor het eerst op de Passo Albetone in
Italië: gewoon een stuk focaccia met tomaat, mozzarella en oregano, warm van
de grill. Verder niets bijzonders, maar met rijpe tomaten en romige
mozzarella is het precies genoeg. Heb je wat meer tijd, bak dan zelf een
focaccia (zie blz.~\pageref{recipe:focaccia}); een kant-en-klare werkt net zo
goed als het wat sneller moet.

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{2 bollen}{mozzarella (bij voorkeur romige buffelmozzarella)}\index[register]{Mozzarella}
  \ing{1}{focaccia (kant-en-klaar, of zelfgebakken)}\index[register]{Focaccia}
  \ing{5}{rijpe pruimtomaten, ontdaan van de zaadlijsten}\index[register]{Tomaat}
  \ing{1 tl}{gedroogde oregano}
  \ing{2 el}{extra vergine olijfolie}
  \ingb{zout en versgemalen zwarte peper, naar smaak}
\end{ingredients}

\blockrule{Bereiding}
\tip{Laat de tomaat na het verwijderen van de zaadlijsten even uitlekken op
keukenpapier: hoe minder vocht erin blijft zitten, hoe minder de panini
doorweekt op de grill.}
\begin{steps}
  \item Verwarm de contactgrill voor op de hoogste stand.
  \item Snijd de focaccia in 5 gelijke stukken en snijd elk stuk open.
  \item Was de tomaten, halveer ze en verwijder de zaadlijsten met een
        lepeltje.
  \item Snijd de mozzarella in plakken.
  \item Beleg de onderste helft van elk stuk focaccia met plakken mozzarella
        en tomaat, bestrooi met de oregano, het zout en de peper, en
        besprenkel met de olijfolie.
  \item Leg de bovenste helft van de focaccia erop en druk lichtjes aan.
  \item Leg de panini's in de contactgrill en pers dicht. Grill 3 tot 4
        minuten, tot het brood goudbruin is en de mozzarella gesmolten is.
  \item Serveer direct, warm uit de grill.
\end{steps}

\end{recipe}
