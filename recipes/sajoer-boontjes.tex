\begin{recipe}{Sajoer boontjes}
\kicker[Bijgerecht,Vegetarisch,Indonesisch]{Bijgerecht · vegetarisch · Indonesisch}
\index[register]{Sajoer boontjes}
\index[register]{Sperziebonen}

\meta{20 minuten \dvd Eén pan}

\lettrine{S}{ajoer} boontjes is een Indonesisch groentegerecht met sperziebonen
in een gekruide, licht ingekookte saus. Dit recept gebruikt een boemboe uit
een pakje, wat het een eenvoudig bijgerecht maakt bij bijvoorbeeld nasi of
saté.

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{500 g}{sperziebonen}\index[register]{Sperziebonen}
  \ing{1 el}{zonnebloemolie}
  \ing{45 g}{boemboe sajoer boontjes (b.v.\ Djas)}
  \ing{100 ml}{water}
\end{ingredients}

\blockrule{Bereiding}
\begin{steps}
  \item Was de sperziebonen en haal de eventuele draad en steeltjes eraf.
  \item Kook de sperziebonen 2 tot 3 minuten in ruim kokend water en giet ze af.
  \item Verhit de zonnebloemolie in een wokpan op hoog vuur en roerbak de
        sperziebonen 2 minuten.
  \item Voeg het water en de boemboe sajoer boontjes toe en roer goed door.
  \item Laat het geheel zonder deksel op middelhoog vuur 5 tot 10 minuten
        inkoken, tot de saus is ingedikt.
\end{steps}
\end{recipe}
