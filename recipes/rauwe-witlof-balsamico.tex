\begin{recipe}{Rauwe witlof met crema di balsamico}
\label{recipe:rauwe-witlof-balsamico}
\kicker[Bijgerecht,Vegetarisch,Snack]{Bijgerecht · vegetarisch · snack}
\index[register]{Rauwe witlof met crema di balsamico}
\index[register]{Witlof}
\index[register]{Balsamico}

\meta{10 minuten \dvd Eenvoudig}

\lettrine{R}{auwe} witlof is knapperig en wat bitter, maar met olijfolie en een
streepje zoete crema di balsamico valt dat mee. De kwarten kan je zo uit de hand
eten, daarom is dit bij de kinderen een goed tussendoortje.

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{4}{stronken witlof}\index[register]{Witlof}
  \ingb{olijfolie}
  \ingb{crema di balsamico}\index[register]{Balsamico}
  \ingb{knoflookpoeder, naar smaak}
  \ingb{zout \& versgemalen peper, naar smaak}
\end{ingredients}

\blockrule{Bereiding}
\tip{Crema di balsamico is een dikke, zoete balsamicosaus die je kant en klaar in de supermarkt koopt. Een gewone balsamicoazijn is dunner en zuurder.}
\begin{steps}
  \item Snijd het worteleinde van de witlof en verwijder losse of beschadigde
        buitenblaadjes. Snijd elke stronk in de lengte in vieren.
  \item Snijd de harde kern uit elke kwart, die is extra bitter.
  \item Leg de kwarten naast elkaar op een schaal en besprenkel ze met olijfolie.
  \item Bestrooi met knoflookpoeder, zout en peper en druppel de crema di
        balsamico erover. Serveer direct.
\end{steps}
\end{recipe}
