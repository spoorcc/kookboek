\begin{recipe}{Involtini di maiale}
\kicker[Hoofdgerecht,Vlees,Italiaans]{Hoofdgerecht · vlees · Italiaans}
\index[register]{Involtini di maiale}
\index[register]{Varkensvlees}
\index[register]{Prosciutto}
\index[register]{Kaas}

\meta{30 minuten \dvd Eén pan}
\marginimage{images/involtini-di-maiale-ingredienten}

\lettrine{I}{nvoltini} di maiale zijn dungeslagen lapjes varkensfilet, gevuld
met ham en kaas, strak opgerold en rondom goudbruin gebakken. De pan waarin ze
bakken wordt afgeblust met witte wijn, zodat er direct een glanzende saus bij
zit. Ik maakte dit voor het eerst op de camping, gewoon omdat ik eens iets
anders wilde dan saltimbocca (p.~\pageref{recipe:saltimbocca}). Lekker met
gebakken aardappeltjes en een frisse groene salade
(p.~\pageref{recipe:groene-salade}).

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{750 g}{dunne lapjes varkensfilet (10 stuks)}\index[register]{Varkensvlees}
  \ing{10 plakken}{prosciutto cotto}\index[register]{Prosciutto}
  \ing{125 g}{provolone, in dunne plakjes}\index[register]{Kaas}
  \ing{5 el}{tarwebloem, om te bestuiven}
  \ing{190 ml}{droge witte wijn}\index[register]{Witte wijn}
  \ing{3 takjes}{verse rozemarijn}\index[register]{Rozemarijn}
  \ing{3 teentjes}{knoflook, gehalveerd}\index[register]{Knoflook}
  \ing{3 el}{extra vierge olijfolie}
  \ing{25 g}{ongezouten roomboter}
  \ingb{zout en versgemalen zwarte peper}
  \ingb{houten prikkers}
\end{ingredients}

\blockrule{Bereiding}
\tip{Provolone is lastig te vinden? Scamorza werkt net zo goed en smelt op
dezelfde manier.}
\begin{steps}
  \item Leg de lapjes varkensfilet tussen twee vellen vershoudfolie en sla ze
        met een deegroller of de bodem van een zware pan voorzichtig dun tot
        ca.\ 3 tot 4 mm.
  \item Bestrooi elk lapje met een beetje zout en versgemalen zwarte peper.
        Leg op elk lapje een plak prosciutto cotto en daarop een plakje
        provolone.
  \item Rol het vlees vanaf de korte kant strak op en zet de sluiting vast
        met een of twee houten prikkers, zodat de vulling er niet uitloopt.
  \item Wentel de rolletjes licht door de tarwebloem en schud het
        overtollige bloem eraf.
  \item Verhit de olijfolie en de roomboter in een grote koekenpan op
        middelhoog vuur. Bak de involtini rondom goudbruin, 2 tot 3 minuten
        per kant.
  \item Voeg de gehalveerde knoflook en de rozemarijn toe aan de pan en bak
        1 minuut mee, tot ze geurig zijn.
  \item Giet de witte wijn in de pan en laat op hoog vuur in 1 tot 2 minuten
        grotendeels verdampen.
  \item Zet het vuur laag, leg het deksel schuin op de pan en laat nog 3 tot
        5 minuten zachtjes stoven, tot er een romige, glanzende saus is
        ontstaan.
  \item Verwijder de prikkers en schep de pansaus over de involtini. Serveer
        direct.
\end{steps}

\end{recipe}
