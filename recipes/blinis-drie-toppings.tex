\begin{recipe}{Blini's met drie toppings}
\kicker[Voorgerecht,Snack]{Voorgerecht · snack}
\index[register]{Blini's met drie toppings}
\index[register]{Blini's}
\index[register]{Gerookte zalm}
\index[register]{Rosbief}
\index[register]{Geitenkaas}

\meta{15 minuten \dvd 6 stuks}

\lettrine{B}{lini's} zijn kleine, dikke pannenkoekjes en een handig hapje voor
bij de borrel: je belegt ze koud, dus er komt geen kookwerk meer bij. Deze
drie toppings, met zalm, met rosbief en een vegetarische met geitenkaas,
maken samen een gevarieerd plankje.

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{6 stuks}{blini's}\index[register]{Blini's}
\end{ingredients}

\medskip
\noindent\textit{Met gerookte zalm}
\begin{ingredients}
  \ing{2 plakjes}{gerookte zalm}\index[register]{Gerookte zalm}
  \ing{2 tl}{kruidenroomkaas (bijv.\ Boursin)}\index[register]{Roomkaas}
  \ing{3 takjes}{bieslook, fijngesneden}\index[register]{Bieslook}
\end{ingredients}

\medskip
\noindent\textit{Met rosbief}
\begin{ingredients}
  \ing{2 plakjes}{rosbief (met peper of naturel)}\index[register]{Rosbief}
  \ing{10 g}{rucola}\index[register]{Rucola}
  \ing{2 tl}{groene pesto}\index[register]{Pesto}
  \ing{2 tl}{Parmezaanse kaas, geraspt}\index[register]{Parmezaanse kaas}
\end{ingredients}

\medskip
\noindent\textit{Met geitenkaas, zongedroogde tomaat en honing}
\begin{ingredients}
  \ing{2 stuks}{zachte geitenkaas, in schijfjes}\index[register]{Geitenkaas}
  \ing{2 stuks}{zongedroogde tomaat (op olie)}\index[register]{Zongedroogde tomaten}
  \ing{2 tl}{vloeibare honing}
  \ing{1 takje}{verse tijm}
\end{ingredients}

\blockrule{Bereiding}
\tip{Maak de drie toppings vlak voor het serveren op: de blini's worden
anders snel zacht.}
\begin{steps}
  \item Leg de blini's op een bord of plank.
  \item Voor de zalmvariant: schep op twee blini's een beetje kruidenroomkaas
        en leg daar een plakje gerookte zalm overheen. Werk af met de
        fijngesneden bieslook.
  \item Voor de rosbiefvariant: beleg twee blini's met een beetje rucola en
        leg daar een plakje rosbief overheen. Schep er een theelepel pesto
        op en werk af met de Parmezaanse kaas.
  \item Voor de geitenkaasvariant: leg op twee blini's een schijfje geitenkaas
        en daarop een stukje zongedroogde tomaat, in reepjes gesneden of
        in zijn geheel. Werk af met een toefje honing en het takje tijm.
  \item Serveer direct.
\end{steps}
\end{recipe}
