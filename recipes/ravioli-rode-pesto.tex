\begin{recipe}{Ravioli met rode pesto}
\kicker[Hoofdgerecht,Vegetarisch,Italiaans,Doordeweeks]{Hoofdgerecht · vegetarisch · Italiaans · doordeweeks}
\index[register]{Ravioli met rode pesto}
\index[register]{Ravioli}
\index[register]{Pesto, rode}
\index[register]{Spinazie}
\index[register]{Parmezaanse kaas}

\meta{20 minuten}

\lettrine{E}{en} snelle romige pastasaus van rode pesto, kookroom en spinazie,
met gevulde ravioli erdoorheen. Weinig werk voor veel smaak, en handig voor
een doordeweekse avond.

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{200 g}{rode pesto}\index[register]{Pesto, rode}
  \ing{750 g}{verse ravioli met kaasvulling}\index[register]{Ravioli}
  \ing{200 g}{verse spinazie}\index[register]{Spinazie}
  \ing{200 ml}{kookroom}\index[register]{Kookroom}
  \ing{50 g}{Parmezaanse kaas, geraspt}\index[register]{Parmezaanse kaas}
  \ing{1}{ui}\index[register]{Ui}
  \ingb{olijfolie}
  \ingb{chilivlokken}
  \ingb{zout en versgemalen peper}
  \ingb{extra Parmezaanse kaas, om te serveren}
\end{ingredients}

\blockrule{Bereiding}
\tip{Kan ook met tortellini of een andere gevulde pasta.}
\begin{steps}
  \item Zet een grote pan water met zout op voor de ravioli.
  \item Snipper de ui.
  \item Rooster de chilivlokken in een droge hapjespan op middelhoog vuur, 30
        tot 60 seconden, tot ze geurig zijn.
  \item Voeg een scheutje olijfolie en de ui toe en bak op laag vuur tot de ui
        glazig is, ca.\ 5 minuten.
  \item Voeg de rode pesto toe en laat een minuutje meebakken.
  \item Voeg de kookroom toe en daarna de spinazie in delen; laat op laag vuur
        slinken tot alle spinazie erdoor is.
  \item Kook de ravioli volgens de verpakking gaar, meestal 2 tot 3 minuten.
  \item Schep de ravioli met een scheutje kookvocht en de Parmezaanse kaas
        door de saus. Voeg het kookvocht met mate toe: de spinazie laat zelf
        ook vocht los, dus de saus wordt snel te nat.
  \item Breng op smaak met ruim peper en zout.
  \item Serveer direct met extra Parmezaanse kaas erover.
\end{steps}
\end{recipe}
