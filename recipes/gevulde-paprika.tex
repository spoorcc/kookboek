\begin{recipe}{Gevulde paprika}
\kicker[Hoofdgerecht,Vlees,Doordeweeks]{Hoofdgerecht · vlees · doordeweeks}
\index[register]{Gevulde paprika}
\index[register]{Paprika}
\index[register]{Rundergehakt}
\index[register]{Kikkererwten}
\index[register]{Spinazie}

\meta{50 minuten \dvd Oven 200\,°C}

\lettrine{G}{evulde} paprika's zijn een stevige ovenschotel met gehakt,
kikkererwten en spinazie, op smaak gebracht met een flinke scheut komijn,
koriander en paprikapoeder. De rijst en de kikkererwten maken de vulling
vullend genoeg voor een avondmaaltijd, en in de oven wordt de paprika zelf
mooi zacht terwijl de kaas erboven smelt.

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{5 grote}{paprika's}\index[register]{Paprika}
  \ing{400 g}{rundergehakt}\index[register]{Rundergehakt}
  \ing{400 g}{kikkererwten (1 blik), uitgelekt}\index[register]{Kikkererwten}
  \ing{150 g}{gekookte rijst}\index[register]{Rijst}
  \ing{400 g}{verse spinazie}\index[register]{Spinazie}
  \ing{1,5}{ui}
  \ing{2 teentjes}{knoflook}
  \ing{2 tl}{komijnpoeder}
  \ing{1,5 tl}{paprikapoeder (gerookt of mild)}
  \ing{1 tl}{korianderpoeder (ketoembar)}
  \ing{0,5 tl}{chilivlokken of cayennepeper (optioneel, voor wat pit)}
  \ing{1}{runderbouillonblokje}
  \ing{100 g}{geraspte belegen kaas}\index[register]{Kaas}
  \ingb{olijfolie of boter, om te bakken}
\end{ingredients}

\blockrule{Bereiding}
\tip{Blijft er vulling over? Bak die apart mee in een klein ovenschaaltje, of
gebruik hem de volgende dag als vulling voor een wrap.}
\begin{steps}
  \item Verwarm de oven voor op 200\,°C (boven- en onderwarmte).
  \item Was de paprika's, snijd de kapjes eraf en verwijder de zaadlijsten.
  \item Snipper de ui en pers de knoflook.
  \item Verhit een scheut olijfolie of boter in een grote pan en fruit de ui
        en knoflook op middelhoog vuur 3 tot 5 minuten tot de ui glazig is.
  \item Voeg het gehakt toe en bak dit op middelhoog tot hoog vuur rul, tot
        het rondom bruin is.
  \item Roer de komijn, paprikapoeder, korianderpoeder en eventueel de
        chilivlokken of cayennepeper erdoor en bak de kruiden 1 minuut mee
        tot ze geurig worden.
  \item Verkruimel het bouillonblokje boven de pan en roer goed door.
  \item Voeg de spinazie in delen toe en laat op middelhoog vuur slinken tot
        alle spinazie is geslonken.
  \item Haal de pan van het vuur en meng de uitgelekte kikkererwten en de
        gekookte rijst erdoor.
  \item Vul de paprika's stevig met het mengsel en zet ze rechtop naast
        elkaar in een ovenschaal.
  \item Strooi de geraspte kaas over de gevulde paprika's.
  \item Bak de paprika's 25 tot 30 minuten in de oven, tot de kaas goudbruin
        en gesmolten is en de paprika's zacht zijn.
\end{steps}
\end{recipe}
