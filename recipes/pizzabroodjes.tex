\begin{recipe}{Pizzabroodjes}
\kicker[Hoofdgerecht,Vegetarisch,Italiaans,Lunch]{Hoofdgerecht · vegetarisch · Italiaans · lunch}
\index[register]{Pizzabroodjes}
\index[register]{Mozzarella}
\index[register]{Tomaat}
\index[register]{Brood}

\meta{35 minuten \dvd Voor 2 personen \dvd Eén pan}

\lettrine{D}{eze} pizzabroodjes maken we vaak op de camping, als het brood van
de dag ervoor te droog is om zomaar nog op te eten. Een snelle tomatensaus,
een plak brood en wat mozzarella erover, en het oudbakken brood krijgt alsnog
een goede bestemming.

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{1 bol (125 g)}{mozzarella, in stukjes gescheurd}\index[register]{Mozzarella}
  \ing{6 plakken}{oudbakken brood}\index[register]{Brood}
  \ing{250 g}{rijpe tomaten, in kleine stukjes}\index[register]{Tomaat}
  \ing{1 teen}{knoflook, gehalveerd}
  \ingb{olijfolie, voor het bakken en afwerken}
  \ingb{gedroogde oregano}
  \ingb{verse basilicum, naar wens}
  \ingb{zout en versgemalen zwarte peper, naar smaak}
\end{ingredients}

\blockrule{Bereiding}
\tip{Gebruik een koekenpan met deksel: de deksel houdt de warmte vast zodat de
mozzarella smelt zonder dat de onderkant van het brood verbrandt.}
\begin{steps}
  \item Was de tomaten en snijd ze in kleine stukjes.
  \item Verhit een scheut olijfolie in een koekenpan en bak het gehalveerde
        teentje knoflook op laag vuur tot het net niet bruin kleurt, ongeveer
        2 minuten. Voeg de tomaten toe en laat het geheel, met de deksel op de
        pan, op laag vuur 10 tot 15 minuten sudderen tot een dik sausje. Breng
        op smaak met zout en peper en verwijder het teentje knoflook.
  \item Snijd het brood in plakken en besprenkel beide zijden met olijfolie.
  \item Bedek één zijde van elke plak met de tomatensaus en verdeel de
        mozzarella erover.
  \item Leg de broodjes met de belegde zijde omhoog in de koekenpan en bak op
        laag vuur 4 tot 5 minuten tot de onderkant knapperig is. Leg dan de
        deksel op de pan en bak nog 2 tot 3 minuten tot de mozzarella
        gesmolten is.
  \item Haal van het vuur, druppel er wat olijfolie over en bestrooi met de
        oregano en, naar wens, wat verse basilicum. Serveer direct.
\end{steps}
\end{recipe}
