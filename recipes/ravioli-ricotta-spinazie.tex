\begin{recipe}{Zelfgemaakte ravioli met ricotta en spinazie}
\label{recipe:ravioli-ricotta-spinazie}
\kicker[Hoofdgerecht,Vegetarisch,Italiaans]{Hoofdgerecht · vegetarisch · Italiaans}
\index[register]{Zelfgemaakte ravioli met ricotta en spinazie}
\index[register]{Ravioli}
\index[register]{Ricotta}
\index[register]{Spinazie}

\meta{190 minuten \dvd Raviolimaker}
\marginimage{images/ravioli-ricotta-spinazie-ingredienten}

\lettrine{V}{oor} mijn verjaardag kreeg ik een raviolimaker, en dit was het eerste
recept dat ik ermee probeerde. Ricotta met spinazie is een klassieke vulling om
mee te beginnen. Het deeg komt uit de handleiding en levert twaalf ravioli per
keer op. Zelf ricotta maken kan ook, zie p.~\pageref{recipe:ricotta}.

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{100 g}{ricotta}\index[register]{Ricotta}
  \ing{200 g}{verse spinazie}\index[register]{Spinazie}
  \ing{3}{eieren}\index[register]{Eieren}
  \ing{150 g}{bloem tipo 00}\index[register]{Bloem}
  \ing{150 g}{semolina (fijn gemalen durum)}
  \ing{50 g}{Parmezaanse kaas, geraspt}\index[register]{Parmezaanse kaas}
  \ing{1}{extra eidooier, om de randen te bestrijken}
  \ing{2 tl}{olijfolie}
  \ingb{extra bloem en semolina, om te bestuiven}
  \ingb{nootmuskaat, zout en versgemalen peper}
\end{ingredients}

\blockrule{Bereiding}
\tip{Knijp de spinazie goed uit, anders gaan de ravioli open. Lekker met boter
of met de saus van p.~\pageref{recipe:ravioli-rode-pesto}.}
\begin{steps}
  \item Maak een kuiltje in de bloem en de semolina op een schone werkbank.
        Voeg de eieren, de olie, een scheutje water en een snufje zout en peper
        toe, meng met een vork en kneed tot een gladde bal. Wikkel in
        vershoudfolie en laat minimaal 2 tot 3 uur rusten in de koelkast, het
        liefst een nacht.
  \item Was de spinazie. Bak hem in een scheutje olijfolie op middelhoog vuur
        1 minuut tot hij begint te slinken. Zet het vuur laag en laat onder
        een deksel 5 tot 6 minuten garen. Breng op smaak met zout.
  \item Druk de spinazie goed uit in een vergiet en hak hem fijn. Meng met de
        ricotta, de Parmezaanse kaas, peper en nootmuskaat.
  \item Halveer het deeg en houd de ene helft afgedekt. Bestuif de andere helft
        met semolina, rol hem uit tot een rechthoek zo breed als de
        pastamachine en haal hem op de breedste stand door de machine. Vouw de
        zijkanten naar het midden, rol hem weer plat en herhaal dit, steeds
        dunner, tot het deeg 1 tot 2 mm dik is. Rol de andere helft op
        dezelfde manier uit.
  \item Snijd de banen iets groter dan de raviolimaker. Restjes kun je opnieuw
        kneden en uitrollen.
  \item Bestuif de raviolimaker royaal met bloem, leg een deegbaan erop en druk
        het deeg met de kunststof vorm in de vakjes. Schep in elk vakje een
        theelepel vulling.
  \item Bestrijk de randen met eidooier, leg de tweede baan erover en rol met
        lichte druk over de maker, zodat de ravioli dichtgedrukt en afgesneden
        worden. Keer de maker om boven een met semolina bestrooide theedoek.
        Herhaal tot al het deeg en de vulling op zijn.
  \item Kook de ravioli in ruim gezouten water op middelhoog vuur. Ze zijn gaar
        zodra ze bovendrijven, na 2 tot 3 minuten.
\end{steps}
