\begin{recipe}{Gnocchi met sperziebonen en pesto}
\kicker[Hoofdgerecht,Vegetarisch,Italiaans]{Hoofdgerecht · vegetarisch · Italiaans}
\index[register]{Gnocchi met sperziebonen en pesto}
\index[register]{Gnocchi}
\index[register]{Sperziebonen}
\index[register]{Pesto, verse}
\index[register]{Parmezaanse kaas}
\index[register]{Pijnboompitten}

\meta{20 minuten}

\marginimage{images/gnocchi-sperziebonen-pesto}
\lettrine{E}{en} klassieke combinatie uit Ligurië: gnocchi, sperziebonen en
pesto uit dezelfde pan, zoals ze dat in Genua ook doen. Ook fijn voor op de
camping: houd pesto, kaas en gnocchi gekoeld, al is dat minder kritisch dan
bij vlees.

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{400 g}{sperziebonen, afgehaald}\index[register]{Sperziebonen}
  \ing{170 g}{verse basilicumpesto}\index[register]{Pesto, verse}
  \ing{800 g}{verse gnocchi}\index[register]{Gnocchi}
  \ing{50 g}{pijnboompitten}\index[register]{Pijnboompitten}
  \ingb{Parmezaanse kaas, geraspt}\index[register]{Parmezaanse kaas}
\end{ingredients}

\blockrule{Bereiding}
\tip{Past niet alles tegelijk in de pan? Kook de sperziebonen en gnocchi dan
in twee porties, in hetzelfde water. Heb je tijd over? Maak de pesto dan
zelf (p.~\pageref{recipe:pesto-genovese}).}
\begin{steps}
  \item Was de sperziebonen en kook ze 7 minuten alleen in een grote pan
        gezouten water op hoog vuur: zo blijven ze knapperig, een fijn
        contrast met de gnocchi.
  \item Rooster ondertussen de pijnboompitten zonder vet in een steelpannetje
        op middelhoog vuur, 2 tot 3 minuten tot ze goudbruin en geurig zijn.
        Haal ze direct uit de pan.
  \item Voeg de gnocchi toe aan de pan met de sperziebonen en haal hem na 2
        minuten van het vuur: dan drijven de gnocchi boven en zijn ze precies
        gaar.
  \item Schep ongeveer 60 ml van het kookvocht apart voor je afgiet: een
        scheutje in de saus maakt de pesto luchtiger, te veel juist waterig.
  \item Giet gnocchi en bonen af en schep ze in een warme kom. Roer de
        pesto, een deel van de kaas en het bewaarde kookvocht erdoor tot een
        gladde, glanzende saus.
  \item Verdeel over de borden en serveer met extra kaas, de pijnboompitten
        en brood om de laatste restjes pesto op te vegen. De pesto heeft
        geen extra olie nodig.
\end{steps}
\end{recipe}
