\begin{recipe}{Risi e bisi}
\kicker[Hoofdgerecht,Vlees,Venetiaans]{Hoofdgerecht · vlees · Venetiaans}
\index[register]{Risi e bisi}
\index[register]{Rijst}
\index[register]{Erwtjes}
\index[register]{Pancetta}

\meta{25 minuten \dvd Voor 4 porties \dvd Eén pan}

\lettrine{R}{isi e bisi} komt uit Venetië en zit gek genoeg tussen risotto en
soep in: losser dan een klassieke risotto, dikker dan een soep, en je eet het
met een lepel. Doordat de rijst niet eerst wordt gefruit, geeft hij meteen
zijn zetmeel af aan de bouillon en sta je binnen een half uur met één pan
warme, romige rijst op tafel.

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{300 g}{risottorijst (Arborio of Vialone Nano)}\index[register]{Rijst}
  \ing{400 g}{diepvrieserwtjes}\index[register]{Erwtjes}
  \ing{50 g}{pancetta, fijngesneden}\index[register]{Pancetta}
  \ing{1}{kleine ui, gesnipperd}
  \ing{1,2 liter}{warme kippenbouillon}
  \ing{50 g}{ongezouten roomboter}
  \ing{50 g}{Parmigiano Reggiano, geraspt}\index[register]{Parmezaanse kaas}
  \ingb{scheutje olijfolie, om te fruiten}
\end{ingredients}

\blockrule{Bereiding}
\tip{Risi e bisi is van oorsprong dunner dan een risotto, bijna een dikke
soep die je met een lepel eet. Wordt hij te dik terwijl hij rust, roer er dan
nog een scheutje warme bouillon door.}
\begin{steps}
  \item Verhit 25 g van de boter met een scheutje olijfolie in de hapjespan
        op middelhoog vuur. Fruit de ui en de pancetta 3 minuten, tot de ui
        glazig is.
  \item Giet de warme kippenbouillon erbij en breng aan de kook.
  \item Voeg de risottorijst direct aan de kokende bouillon toe, zonder hem
        eerst mee te bakken: zo geven de korrels meteen hun zetmeel af aan de
        soep. Kook de rijst 12 minuten zachtjes, af en toe roerend.
  \item Voeg de diepvrieserwtjes toe en kook nog 3 tot 5 minuten mee, tot de
        rijst beetgaar is en de erwtjes warm maar nog felgroen zijn.
  \item Haal de pan van het vuur. Roer de resterende 25 g koude boter en de
        Parmigiano Reggiano erdoor tot een romige, dikke soep ontstaat.
  \item Laat 2 minuten afgedekt rusten en serveer direct.
\end{steps}
\end{recipe}
