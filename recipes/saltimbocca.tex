\begin{recipe}{Saltimbocca alla Romana}
\kicker[Hoofdgerecht,Vlees,Italiaans]{Hoofdgerecht · vlees · Italiaans}
\index[register]{Saltimbocca alla Romana}
\index[register]{Kalfsvlees}
\index[register]{Prosciutto}
\index[register]{Salie}

\meta{30 minuten \dvd Eén pan}

\lettrine{S}{altimbocca} alla Romana is een Romeinse klassieker: dungeslagen
kalfsvlees met prosciutto en salie, kort gebakken en afgeblust met witte wijn
tot een glanzende jus. Het is in een halfuur op tafel en past goed bij
gnocchi met salie-roomboter (p.~\pageref{recipe:gnocchi-salie-roomboter}) en
een frisse groene salade (p.~\pageref{recipe:groene-salade}).

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{5}{grote kalfsschnitzels (ca.\ 500--625 g in totaal)}\index[register]{Kalfsvlees}
  \ing{5 tot 10 plakken}{prosciutto}\index[register]{Prosciutto}
  \ing{10 blaadjes}{verse salie}\index[register]{Salie}
  \ing{60 g}{ongezouten roomboter, in 2 porties}
  \ing{125 ml}{droge witte wijn (bijv.\ Pinot grigio)}\index[register]{Witte wijn}
  \ing{4 el}{tarwebloem, om te bloemen}
  \ingb{versgemalen zwarte peper}
  \ingb{olijfolie, om te bakken}
  \ingb{houten prikkers}
\end{ingredients}

\blockrule{Bereiding}
\tip{Zodra de prosciutto in de pan geurt, is het vlees bijna klaar: kalfsvlees
zo dun geslagen gaart razendsnel en is binnen een paar minuten droog als je
het te lang laat liggen.}
\begin{steps}
  \item Leg de kalfsschnitzels tussen vershoudfolie en sla ze met een
        vleeshamer of de bodem van een zware pan heel dun, tot ca.\ 3 tot
        4 mm. Snijd ze eventueel in gelijke, handzame stukken.
  \item Bestrooi het vlees met wat versgemalen peper. Leg op elk stuk een
        plak prosciutto en daarbovenop een blaadje salie.
  \item Rijg een houten prikker dwars door de salie en de prosciutto recht
        in het vlees, zodat alles goed vastzit.
  \item Strooi de tarwebloem op een bord. Druk alleen de kale kant van het
        vlees kort in de bloem en schud het overtollige bloem eraf.
  \item Verhit een scheut olijfolie met de helft van de roomboter in een
        grote koekenpan op middelhoog vuur. Leg het vlees met de
        prosciuttokant naar beneden in de pan en bak ca.\ 1,5 minuut, tot de
        ham geurt. Draai om en bak de vleeskant nog hooguit 1 minuut. Leg
        het vlees direct op een warm bord en dek af met aluminiumfolie.
  \item Giet de witte wijn in de hete pan en schraap de aanbaksels los. Laat
        op hoog vuur tot de helft inkoken, ca.\ 2 minuten. Zet het vuur laag
        en roer de resterende koude roomboter erdoor tot een glanzende jus.
  \item Verwijder de prikkers en lepel de warme jus over het kalfsvlees.
        Serveer direct.
\end{steps}
