\begin{recipe}{Geroerbakte spruitjes}
\kicker[Bijgerecht,Vlees,Nederlands,Doordeweeks]{Bijgerecht · vlees · Nederlands · doordeweeks}
\index[register]{Geroerbakte spruitjes}
\index[register]{Spruitjes}
\index[register]{Spek}
\index[register]{Rode ui}

\meta{25 minuten \dvd Eén pan}

\lettrine{G}{eroerbakte} spruitjes met spekjes en rode ui zijn een snel
bijgerecht dat goed past bij een stevige aardappelmaaltijd. De spruitjes
worden eerst kort geblancheerd en daarna samen met het spek en de ui
geroerbakt, zodat ze nog wat bite houden.

\blockrule{Ingrediënten}
\begin{ingredients}
  \ing{500 g}{spruitjes}\index[register]{Spruitjes}
  \ing{150 g}{spekblokjes}\index[register]{Spek}
  \ing{1}{rode ui}\index[register]{Rode ui}
  \ingb{versgemalen peper en zout}
\end{ingredients}

\blockrule{Bereiding}
\begin{steps}
  \item Snipper de rode ui. Maak de spruitjes schoon: verwijder de buitenste
        blaadjes en snijd de onderkant bij.
  \item Blancheer de spruitjes 4 tot 5 minuten in ruim kokend water tot ze
        beetgaar zijn en giet ze af.
  \item Bak de spekblokjes op middelhoog vuur in een droge koekenpan tot ze
        krokant zijn.
  \item Voeg de rode ui toe en bak op middelhoog vuur 3 tot 4 minuten tot
        hij glazig is.
  \item Voeg de spruitjes toe en roerbak op middelhoog tot hoog vuur 3 tot
        4 minuten mee tot ze warm zijn en een beetje kleur hebben.
  \item Breng op smaak met peper en zout.
\end{steps}
\end{recipe}
